package app

import (
	"bytes"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/DingTalk-Real-AI/dingtalk-workspace-cli/internal/i18n"
	"github.com/DingTalk-Real-AI/dingtalk-workspace-cli/internal/testseam"
	"github.com/fatih/color"
	"github.com/spf13/cobra"
)

func TestCrossPlatformCoverageRootHelpProjectionOracle(t *testing.T) {
	directory := os.Getenv("DWS_HELP_RENDER_ORACLE")
	if directory == "" {
		t.Skip("explicit local compatibility capture")
	}
	if err := os.MkdirAll(directory, 0700); err != nil {
		t.Fatal(err)
	}
	t.Setenv("DWS_CONFIG_DIR", filepath.Join(t.TempDir(), "config"))
	previous := i18n.Lang()
	t.Cleanup(func() { i18n.SetLang(previous) })
	for _, locale := range []string{"en", "zh"} {
		for _, colored := range []bool{false, true} {
			for _, shape := range []string{"full", "source", "empty", "custom"} {
				t.Run(fmt.Sprintf("%s-%t-%s", locale, colored, shape), func(t *testing.T) {
					i18n.SetLang(locale)
					testseam.Swap(t, &color.NoColor, !colored)
					root := &cobra.Command{Use: "dws", Long: "  root guidance  "}
					if shape == "full" {
						root = NewRootCommand()
					} else if shape == "source" {
						root = NewSchemaSourceRootCommand()
					} else {
						if shape == "custom" {
							root.AddCommand(&cobra.Command{Use: "calendar", Short: " Calendar details "}, &cobra.Command{Use: "custom", Short: " Utility details "}, &cobra.Command{Use: "secret", Hidden: true})
							root.PersistentFlags().BoolP("yes", "y", false, " confirm action ")
							root.PersistentFlags().String("name", "", " display name ")
							root.PersistentFlags().Int64P("limit", "n", 10, " limit values ")
							root.PersistentFlags().String("hidden", "", "hidden flag")
							if err := root.PersistentFlags().MarkHidden("hidden"); err != nil {
								t.Fatal(err)
							}
						}
						configureRootHelp(root)
					}
					var out, stderr bytes.Buffer
					root.SetOut(&out)
					root.SetErr(&stderr)
					root.SetArgs([]string{"--help"})
					if err := root.Execute(); err != nil || stderr.Len() != 0 {
						t.Fatalf("help execution: %v %s", err, stderr.String())
					}
					if colored != strings.Contains(out.String(), "\x1b[") {
						t.Fatal("color mode was not exercised")
					}
					if err := os.WriteFile(filepath.Join(directory, t.Name()[strings.LastIndex(t.Name(), "/")+1:]+".txt"), out.Bytes(), 0600); err != nil {
						t.Fatal(err)
					}
				})
			}
		}
	}
}
